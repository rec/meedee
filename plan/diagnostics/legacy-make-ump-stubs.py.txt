from pathlib import Path
import re
source=Path('crates/middi/src/ump_values.rs').read_text()
result=['"""Pure UMP words and immutable MIDI 2.0 values; no transport initialization."""','from enum import Enum','from . import messages','']
for name,body in re.findall(r'(\w+) \[0x[0-9a-f]+,0\] \{([^}]*)\};',source):
    fields=re.findall(r'(\w+):\w+ =',body)
    result.append(f'class {name}(messages.Message):')
    params=', '.join(f'{f}: int' for f in fields)
    result.append('    def __init__(self'+(', *, '+params if fields else '')+') -> None: ...')
    for field in fields:result.extend(['    @property',f'    def {field}(self) -> int: ...'])
    result.append(f'    def replace(self, **changes: object) -> {name}: ...\n')
Path('middi/ump.pyi').write_text('\n'.join(result)+'\n')
