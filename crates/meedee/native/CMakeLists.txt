cmake_minimum_required(VERSION 3.22)
project(meedee_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(APPLE AND CMAKE_OSX_DEPLOYMENT_TARGET VERSION_LESS "11.0")
  message(FATAL_ERROR "meedee native transport requires macOS 11 or later")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(feature TESTS EXAMPLES FIND_BOOST DOWNLOAD_CPPWINRT)
  set(LIBREMIDI_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(backend JACK PIPEWIRE NETWORK KEYBOARD WINUWP WINMIDI KDMAPI)
  set(LIBREMIDI_NO_${backend} ON CACHE BOOL "" FORCE)
endforeach()
set(LIBREMIDI_NO_BOOST ON CACHE BOOL "" FORCE)
add_compile_definitions(LIBREMIDI_NO_BOOST_VARIANT2)
add_subdirectory("${MEEDEE_VENDOR}" libremidi)
get_target_property(backends libremidi INTERFACE_COMPILE_DEFINITIONS)
if(APPLE AND NOT "LIBREMIDI_COREMIDI" IN_LIST backends)
  message(FATAL_ERROR "CoreMIDI is required for macOS builds")
elseif(UNIX AND NOT APPLE AND NOT "LIBREMIDI_ALSA" IN_LIST backends)
  message(FATAL_ERROR "Linux builds require ALSA development headers")
elseif(WIN32 AND NOT "LIBREMIDI_WINMM" IN_LIST backends)
  message(FATAL_ERROR "WinMM is required for Windows builds")
endif()
add_library(meedee_native STATIC bridge.cpp)
target_link_libraries(meedee_native PRIVATE libremidi)
if(WIN32)
  target_link_libraries(meedee_native PRIVATE ole32)
endif()
install(TARGETS meedee_native ARCHIVE DESTINATION lib)
